export function TrackCard({ track }: { track: Track }) {
  return (

      )}
    </div>
  );
}

function formatDuration(ms: number): string {
  const total = Math.floor(ms / 1000);
  const m = Math.floor(total / 60);
  const s = total % 60;
  return `${m}:${s.toString().padStart(2, '0')}`;
}

### 5.2 选取逻辑

```typescript
// src/main/services/header-greeting.ts

export async function pickGreeting(): Promise<string> {
  const now = new Date();
  const hour = now.getHours();
  
  // 1. 久别重逢优先(超过 5 天没用)
  const lastUsedAt = await db.getLastUsedAt();
  if (daysSince(lastUsedAt) > 5) {
    return random(HEADER_GREETINGS.long_absence);
  }
  
  // 2. 天气特殊(下雨 / 下雪 / 极热极冷)
  const weather = await getWeather();
  if (weather?.condition.includes('雨')) return random(HEADER_GREETINGS.rain);
  if (weather?.condition.includes('雪')) return random(HEADER_GREETINGS.snow);
  if (weather?.tempC > 32) return random(HEADER_GREETINGS.hot);
  if (weather?.tempC < 0) return random(HEADER_GREETINGS.cold);
  
  // 3. 基于昨日 day_seal(可选 · 30% 概率触发)
  const seal = await daySeal.getYesterday();
  if (seal && Math.random() < 0.3) {
    const fresh = await generateFromSeal(seal);
    if (fresh) return fresh;
  }
  
  // 4. 时间段
  const period = 
    hour < 6  ? 'late_night' :
    hour < 11 ? 'morning' :
    hour < 14 ? 'noon' :
    hour < 17 ? 'afternoon' :
    hour < 19 ? 'evening' :
    hour < 24 ? 'night' : 'late_night';
  
  return random(HEADER_GREETINGS[period]);
}
```

### 5.3 基于昨日 day_seal 的"今日延续"(可选优化)

```typescript
async function generateFromSeal(seal: DaySeal): Promise<string | null> {
  // 简化版:不调 LLM,直接关键词匹配
  if (seal.markings?.some(m => m.kind === 'event_ended')) {
    return '昨天那事结束了 · 今天放松点';
  }
  if (seal.summary.includes('焦虑') || seal.summary.includes('累')) {
    return '昨天看你挺累的 · 今天慢慢来';
  }
  if (seal.summary.includes('开心') || seal.summary.includes('好')) {
    return '昨天状态不错 · 今天接着';
  }
  return null;
}
```

注意:这里**不调 LLM** —— 顶栏问候是高频展示,每次调 LLM 浪费 token。基于关键词匹配足够。

### 5.4 刷新策略

- 每次启动 app 时 pick 一次
- 每天跨日时(凌晨 0 点)重新 pick
- 不需要每分钟刷新,**早上"上午好"维持到中午是合理的**
